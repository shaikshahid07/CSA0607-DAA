ALGORITHM WordBreak(Text, Dictionary)
INPUT:
   Text - string of length n
   Dictionary - collection of valid words
OUTPUT:
   TRUE if Text can be completely segmented
   FALSE otherwise
BEGIN
   DP[0] ← TRUE
   FOR i ← 1 TO n DO
      DP[i] ← FALSE
      FOR j ← 0 TO i - 1 DO
         // DEFECT: DP[j] is never checked
         IF substring(Text, j, i) is in Dictionary THEN
            DP[i] ← TRUE
            BREAK
         END IF
      END FOR
   END FOR
   RETURN DP[n]
END

ALGORITHM WordBreak(Text, Dictionary)
INPUT:
   Text - string of length n
   Dictionary - collection of valid words
OUTPUT:
   TRUE if Text can be completely segmented
   FALSE otherwise
BEGIN
   DP[0] ← TRUE
   FOR i ← 1 TO n DO
      DP[i] ← FALSE
      FOR j ← 0 TO i - 1 DO
         // CORRECTED LINE: DP[j] = TRUE is now required
         IF DP[j] = TRUE AND substring(Text, j, i) is in Dictionary THEN
            DP[i] ← TRUE
            BREAK
         END IF
      END FOR
   END FOR
   RETURN DP[n]
END
