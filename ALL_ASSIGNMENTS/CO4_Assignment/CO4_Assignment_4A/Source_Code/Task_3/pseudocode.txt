ALGORITHM Knapsack01(Weight, Value, n, W)
INPUT:
   Weight[1..n], Value[1..n] - weight and value of each indivisible item
   n - number of items,  W - vehicle capacity
OUTPUT:
   BestValue - maximum total value with total weight <= W
   Chosen    - the set of items that achieves BestValue
BEGIN
   FOR w ← 0 TO W DO   // boundary: no items considered
      DP[0][w] ← 0
   END FOR
   FOR i ← 1 TO n DO   // item index increases: DP[i-1][*] is ready
      FOR w ← 0 TO W DO
         DP[i][w] ← DP[i-1][w]   // exclude item i
         IF Weight[i] <= w THEN   // item i can be included
            IF Value[i] + DP[i-1][w - Weight[i]] > DP[i][w] THEN
               DP[i][w] ← Value[i] + DP[i-1][w - Weight[i]]   // include item i
            END IF
         END IF
      END FOR
   END FOR
   BestValue ← DP[n][W]
   Chosen ← empty set
   w ← W
   FOR i ← n DOWNTO 1 DO   // reconstruction
      IF DP[i][w] != DP[i-1][w] THEN   // value changed, so item i was taken
         ADD i TO Chosen
         w ← w - Weight[i]
      END IF
   END FOR
   RETURN BestValue, Chosen
END
