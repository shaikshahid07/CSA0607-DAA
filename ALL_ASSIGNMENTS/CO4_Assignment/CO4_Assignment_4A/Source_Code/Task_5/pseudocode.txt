ALGORITHM ShortestDistances(V, Edges, Source)
INPUT:
   V       - set of vertices
   Edges   - directed edges (u, v) with weight w(u, v) >= 0
   Source  - starting vertex
OUTPUT:
   Dist[v] - shortest distance from Source to v
   Pred[v] - predecessor of v on that shortest path
BEGIN
   FOR EACH vertex v IN V DO   // initialisation
      Dist[v] ← INFINITY
      Pred[v] ← NONE
      Done[v] ← FALSE
   END FOR
   Dist[Source] ← 0
   FOR count ← 1 TO |V| DO   // termination: at most |V| selections
      u ← vertex with Done[u] = FALSE having the smallest Dist[u]   // next vertex
      IF u does not exist OR Dist[u] = INFINITY THEN
         BREAK   // remaining vertices are unreachable
      END IF
      Done[u] ← TRUE   // greedy commitment: Dist[u] is final
      FOR EACH edge (u, v) leaving u DO
         IF Done[v] = FALSE AND Dist[u] + w(u, v) < Dist[v] THEN   // relaxation
            Dist[v] ← Dist[u] + w(u, v)   // distance update
            Pred[v] ← u   // predecessor update
         END IF
      END FOR
   END FOR
   RETURN Dist, Pred
END

ALGORITHM BellmanFord(V, Edges, Source)
INPUT:  V, Edges with weights (may be negative), Source
OUTPUT: Dist[], Pred[], NegativeCycle
        (TRUE if a negative cycle is reachable from Source)
BEGIN
   FOR EACH vertex v IN V DO
      Dist[v] ← INFINITY
      Pred[v] ← NONE
   END FOR
   Dist[Source] ← 0
   FOR pass ← 1 TO |V| - 1 DO   // repeated relaxation
      Changed ← FALSE
      FOR EACH edge (u, v) IN Edges DO
         IF Dist[u] != INFINITY AND Dist[u] + w(u, v) < Dist[v] THEN
            Dist[v] ← Dist[u] + w(u, v)
            Pred[v] ← u
            Changed ← TRUE
         END IF
      END FOR
      IF Changed = FALSE THEN
         BREAK   // early stop: nothing can improve
      END IF
   END FOR
   NegativeCycle ← FALSE
   FOR EACH edge (u, v) IN Edges DO   // one more pass (the |V|-th)
      IF Dist[u] != INFINITY AND Dist[u] + w(u, v) < Dist[v] THEN
         NegativeCycle ← TRUE
      END IF
   END FOR
   RETURN Dist, Pred, NegativeCycle
END

ALGORITHM FloydWarshall(n, W)
INPUT:  n vertices numbered 1..n
        W[i][j] = edge weight (INFINITY if no edge, 0 if i = j)
OUTPUT: D[i][j] = shortest distance from i to j for every pair
BEGIN
   D ← copy of W   // D(0)
   FOR k ← 1 TO n DO   // allow vertex k as an intermediate
      FOR i ← 1 TO n DO
         FOR j ← 1 TO n DO
            IF D[i][k] + D[k][j] < D[i][j] THEN   // core state transition
               D[i][j] ← D[i][k] + D[k][j]
            END IF
         END FOR
      END FOR
   END FOR
   RETURN D
END
